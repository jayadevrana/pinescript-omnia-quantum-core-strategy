///////////////////////////////////////////////////////////////////////////////
// 11) 🚀 AGGRESSIVE TRAILING STOP LADDER MANAGEMENT
///////////////////////////////////////////////////////////////////////////////
var string currentLevel = "BASE"

if not na(entryPrice) and not na(initialRisk) and initialRisk > 0
    
    if inLong
        currentR = (close - entryPrice) / initialRisk
        peakR = (highWaterMark - entryPrice) / initialRisk
        
        if peakR > maxRReached
            maxRReached := peakR
        
        // Calculate trailing SL and determine current level
        trailingSL = initialSL
        
        if peakR >= level4_r
            trailingSL := entryPrice + (initialRisk * level3_r)
            currentLevel := "TP4"
        else if peakR >= level3_r
            trailingSL := entryPrice + (initialRisk * level2_r)
            currentLevel := "TP3"
        else if peakR >= level2_r
            trailingSL := entryPrice + (initialRisk * level1_r)
            currentLevel := "TP2"
        else if peakR >= level1_r
            trailingSL := entryPrice
            currentLevel := "TP1 (BE)"
        else
            trailingSL := initialSL
            currentLevel := "BASE"
        
        strategy.exit("TRAIL-L", "LONG", stop=trailingSL)
        
        // Pullback exit
        if not na(highWaterMark) and peakR >= level1_r
            pullbackFromHigh = (highWaterMark - close) / highWaterMark * 100
            if pullbackFromHigh >= pullbackPct and barsSincePeak >= pullbackBars
                exitLevel := "PB"
                strategy.close("LONG", comment="Pullback Exit")
        
        // ADX weakness exit
        if useADXExit and currentR >= level2_r and adxVal < adxExitThreshold
            exitLevel := "ADX"
            strategy.close("LONG", comment="Weak ADX")
    
    if inShort
        currentR = (entryPrice - close) / initialRisk
        peakR = (entryPrice - lowWaterMark) / initialRisk
        
        if peakR > maxRReached
            maxRReached := peakR
        
        // Calculate trailing SL and determine current level
        trailingSL = initialSL
        
        if peakR >= level4_r
            trailingSL := entryPrice - (initialRisk * level3_r)
            currentLevel := "TP4"
        else if peakR >= level3_r
            trailingSL := entryPrice - (initialRisk * level2_r)
            currentLevel := "TP3"
        else if peakR >= level2_r
            trailingSL := entryPrice - (initialRisk * level1_r)
            currentLevel := "TP2"
        else if peakR >= level1_r
            trailingSL := entryPrice
            currentLevel := "TP1 (BE)"
        else
            trailingSL := initialSL
            currentLevel := "BASE"
        
        strategy.exit("TRAIL-S", "SHORT", stop=trailingSL)
        
        // Pullback exit
        if not na(lowWaterMark) and peakR >= level1_r
            pullbackFromLow = (close - lowWaterMark) / lowWaterMark * 100
            if pullbackFromLow >= pullbackPct and barsSincePeak >= pullbackBars
                exitLevel := "PB"
                strategy.close("SHORT", comment="Pullback Exit")
        
        // ADX weakness exit
        if useADXExit and currentR >= level2_r and adxVal < adxExitThreshold
            exitLevel := "ADX"
            strategy.close("SHORT", comment="Weak ADX")

        tp4Price = entryPrice - (initialRisk * level4_r)
        if na(tp4Line)
            tp4Line := line.new(entryBarIndex, tp4Price, bar_index, tp4Price, xloc=xloc.bar_index, extend=ext, color=colTP4, style=line.style_dotted, width=lineWidth)
            tp4LineBar := entryBarIndex
            tp4Lab := label.new(bar_index, tp4Price, "TP4 " + str.tostring(level4_r, "#.#") + "R: " + str.tostring(tp4Price, "#.##"), xloc=xloc.bar_index, color=colTP4, textcolor=color.white, style=label.style_label_left, size=size.small)
        else
            line.set_xy2(tp4Line, bar_index, tp4Price)
            label.set_xy(tp4Lab, bar_index, tp4Price)
        
        // Current trailing SL
        peakR = (entryPrice - lowWaterMark) / initialRisk
        currentSL = peakR >= level4_r ? entryPrice - (initialRisk * level3_r) :
                   peakR >= level3_r ? entryPrice - (initialRisk * level2_r) :
                   peakR >= level2_r ? entryPrice - (initialRisk * level1_r) :
                   peakR >= level1_r ? entryPrice : initialSL
        
        if currentSL != initialSL
            if na(currentSLLine)
                currentSLLine := line.new(entryBarIndex, currentSL, bar_index, currentSL, xloc=xloc.bar_index, extend=ext, color=colCurrentSL, style=line.style_solid, width=lineWidth)
                currentSLLineBar := entryBarIndex
                currentSLLab := label.new(bar_index, currentSL, "CURRENT SL: " + str.tostring(currentSL, "#.##"), xloc=xloc.bar_index, color=colCurrentSL, textcolor=color.white, style=label.style_label_left, size=size.small)
            else
                line.set_xy2(currentSLLine, bar_index, currentSL)
                label.set_xy(currentSLLab, bar_index, currentSL)
                label.set_text(currentSLLab, "CURRENT SL: " + str.tostring(currentSL, "#.##"))
        else
            if not na(currentSLLine)
                line.delete(currentSLLine)
                currentSLLine := na
                currentSLLineBar := na
            if not na(currentSLLab)
                label.delete(currentSLLab)
                currentSLLab := na
